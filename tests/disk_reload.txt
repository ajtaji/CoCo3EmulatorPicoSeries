# After tests/disk.txt saved PICO.BAS to the flash disk: a fresh upload and boot still finds it, loads
# and runs it, and saves a second file. Every command waits for its OK before the next is typed
# (keys typed while Disk BASIC is busy are lost, as on the real machine).
expect 15 disks: 1
screen 20 \r\nOK +\r\n
line DIR
screen 10 PICO +BAS +0 +B +1
line LOAD "PICO"
screen 15 LOAD "PICO" +\r\nOK +\r\n
line LIST

line RUN
screen 15 \r\nSAVED ON A PICO +\r\n
line 20 PRINT "AND RELOADED"
line SAVE "TWO"
screen 20 SAVE "TWO" +\r\nOK +\r\n
line DIR
screen 10 TWO +BAS
