# A blank disk in drive 0 (tools/make_disk_region.py --drive 0 blank): a typed program is listed to prove
# it was typed, saved, wiped, loaded back and run; the machine is reset (cold start) and it is loaded from
# the flash again. Every step waits for its result: keys typed while Disk BASIC is busy are lost, as on
# the real machine.
expect 15 disks: 1
screen 20 \r\nOK +\r\n
line DIR
screen 10 DIR +\r\n.*\r\nOK
line 10 PRINT "SAVED ON A PICO"
wait 3
line LIST
screen 10 \r\n10 PRINT "SAVED ON A PICO" +\r\nOK
line SAVE "PICO"
screen 20 SAVE "PICO" +\r\nOK +\r\n
line NEW
wait 1
line LOAD "PICO"
screen 15 LOAD "PICO" +\r\nOK +\r\n
line LIST
screen 10 LIST +\r\n10 PRINT "SAVED ON A PICO" +\r\nOK
line RUN
screen 15 \r\nSAVED ON A PICO +\r\n
cmd r
expect 3 COCO reset
screen 20 DISK EXTENDED COLOR BASIC
screen 10 \r\nOK +\r\n
line LOAD "PICO"
screen 15 LOAD "PICO" +\r\nOK +\r\n
line RUN
screen 15 \r\nSAVED ON A PICO +\r\n
cmd s
expect 3 COCO frames
